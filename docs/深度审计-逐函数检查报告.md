```python
def limit_for(self, window: ClockWindow) -> int:
    """取某窗口的阈值。0 表示该窗口不限流。"""
    if window is None:
        raise ValueError("window 参数不能为 None")
    
    limit = self.windows.get(window.name)
    if limit is not None:
        return limit
    
    # 回退到默认值
    default = DEFAULT_LIMITS.get(window.name, 0)
    
    # 记录回退日志（可选）
    if default > 0 and self.windows:  # 如果配置非空但缺少该窗口
        from fangyu_shared.logging import get_logger
        _logger = get_logger("clock")
        _logger.debug(
            "clock_limit_fallback",
            site_id=self.site_id,
            window=window.name,
            default=default,
            configured_windows=list(self.windows.keys())
        )
    
    return default
```

#### 审计结果

| 检查项 | 状态 | 说明 |
|--------|------|------|
| 输入验证 | ❌ | 未验证 results 长度 |
| 类型安全 | ⚠️ | 假设 results 元素类型 |
| 边界条件 | ❌ | 可能数组越界 |

#### 发现的问题

**问题 1**: 未验证 `results` 长度

```python
    # IndexError!
```

**问题 2**: 假设 `results` 元素可以转 int

```python
int(results[base + 2 + w_idx] or 0)
# 如果 results[x] 是非数值字符串，会抛 ValueError
```

#### 建议修复

```python
def _parse(
    self,
    dims: tuple[tuple[ClockDimension, str], ...],
    results: list,
    now_ms: int,
) -> ClockReading:
    """按固定步长解析 pipeline 结果。"""
    stride = self._ops_per_dimension()
    expected_length = len(dims) * stride
    
    # 验证结果长度
    if len(results) < expected_length:
        _logger.error(
            "clock_parse_insufficient_results",
            expected=expected_length,
            actual=len(results),
            dims_count=len(dims),
            stride=stride
        )
        # 返回空结果，fail-open
        return self._empty_reading(
            dims[0][1] if dims else "", 
            dims[1][1] if len(dims) > 1 else "",
            now_ms
        )
    
    parsed: dict[ClockDimension, DimensionCounts] = {}

    for idx, (dimension, value) in enumerate(dims):
        base = idx * stride
        counts = {}
        
        # 安全解析计数
        for w_idx, window in enumerate(ALL_WINDOWS):
            try:
                raw_count = results[base + 2 + w_idx]
                counts[window.name] = int(raw_count) if raw_count is not None else 0
            except (ValueError, TypeError, IndexError) as e:
                _logger.warning(
                    "clock_parse_count_failed",
                    window=window.name,
                    dimension=dimension.value,
                    error=str(e)
                )
                counts[window.name] = 0
        
        # 安全解析封禁状态
        try:
            ban_raw = results[base + 3 + len(ALL_WINDOWS)]
            ban_ttl = results[base + 4 + len(ALL_WINDOWS)]
        except IndexError:
            _logger.error("clock_parse_ban_index_error", base=base, dimension=dimension.value)
            ban_raw = None
            ban_ttl = 0
        
        parsed[dimension] = DimensionCounts(
            dimension=dimension,
            value=value,
            counts=counts,
            ban=self._parse_ban(ban_raw, ban_ttl),
        )

    return ClockReading(
        ip=parsed.get(ClockDimension.IP) or DimensionCounts(ClockDimension.IP, "", {}, BanState(False)),
        fingerprint=parsed.get(ClockDimension.FINGERPRINT) or DimensionCounts(ClockDimension.FINGERPRINT, "", {}, BanState(False)),
        now_ms=now_ms,
    )
```

```python
@staticmethod
def _parse_ban(raw: object, ttl: object) -> BanState:
    if not raw:
        return BanState(banned=False)
    reason = ""
    try:
        payload = orjson.loads(raw)  # type: ignore[arg-type]
```

```python
@staticmethod
def _parse_ban(raw: object, ttl: object) -> BanState:
    if not raw:
        return BanState(banned=False)
    
    reason = ""
    try:
        payload = orjson.loads(raw)  # type: ignore[arg-type]
        reason = str(payload.get("reason", ""))
    except (orjson.JSONDecodeError, TypeError, ValueError, AttributeError):
        reason = str(raw) if raw else ""
    
    # 更健壮的 TTL 解析
    ttl_int = 0
    try:
        if isinstance(ttl, (int, float)) and ttl > 0:
            ttl_int = int(ttl)
        elif isinstance(ttl, str):
            parsed = int(float(ttl))
            ttl_int = parsed if parsed > 0 else 0
    except (ValueError, TypeError):
        ttl_int = 0
    
    return BanState(banned=True, reason=reason, ttl_seconds=ttl_int)
```

```python
async def ban(
    self,
    site_id: int,
    dimension: ClockDimension,
    value: str,
    *,
    seconds: int,
    reason: str,
) -> None:
    """写入封禁。"""
    
    # 参数验证
    if seconds <= 0:
        return
    
    if site_id < 0:
        _logger.error("clock_ban_invalid_site_id", site_id=site_id)

        _logger.error("clock_ban_empty_value", site_id=site_id, dimension=dimension.value)

    if len(reason) > max_reason_length:

    try:
        await self._redis.set(key, payload, ex=seconds)
```
